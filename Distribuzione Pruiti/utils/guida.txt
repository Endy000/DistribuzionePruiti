CREATE TABLE Utenti (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    Nome VARCHAR(50) NOT NULL,
    Cognome VARCHAR(50) NOT NULL,
    eta INT,
    email VARCHAR(100) UNIQUE,

    token VARCHAR(255)
